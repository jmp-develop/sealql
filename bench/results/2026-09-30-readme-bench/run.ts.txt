type Result = { query: string; term: string; rows: number; plainMs: number; sealSqlMs: number; sealTotalMs: number; ratio: number; mismatch: number };
const output: Result[] = [];
const median = (values: number[]) => [...values].sort((a, b) => a - b)[Math.floor(values.length / 2)];
const ids = (value: any) => value.map((row: any) => String(row.id ?? row.customer?.id ?? row.customerId));
const same = (a: unknown, b: unknown, name: string) => assert.deepEqual(a, b, `${name}: plaintext mismatch`);
async function sealTimed<T>(fn: () => Promise<T>) {
  capture = true; sqlElapsed = 0; const started = performance.now();
  try { const value = await fn(); return { value, sql: sqlElapsed, total: performance.now() - started }; }
  finally { capture = false; }
}
async function plainTimed<T>(fn: () => Promise<T>) { const started = performance.now(); const value = await fn(); return { value, total: performance.now() - started }; }
async function benchmark<T>(name: string, term: string, plainFn: () => Promise<T>, sealFn: () => Promise<T>, signature: (value: T) => unknown, rowCount: (value: T) => number) {
  for (let i = 0; i < 2; i++) {
    const p = await plainTimed(plainFn), e = await sealTimed(sealFn); same(signature(e.value), signature(p.value), name);
  }
  const plain: number[] = [], sealSql: number[] = [], sealTotal: number[] = [];
  let last!: T;
  for (let i = 0; i < 7; i++) {
    let p: Awaited<ReturnType<typeof plainTimed<T>>>, e: Awaited<ReturnType<typeof sealTimed<T>>>;
    if (i % 2 === 0) { p = await plainTimed(plainFn); e = await sealTimed(sealFn); }
    else { e = await sealTimed(sealFn); p = await plainTimed(plainFn); }
    same(signature(e.value), signature(p.value), name); last = e.value;
    plain.push(p.total); sealSql.push(e.sql); sealTotal.push(e.total);
  }
  const plainMs = median(plain), sealSqlMs = median(sealSql), sealTotalMs = median(sealTotal);
  output.push({ query: name, term, rows: rowCount(last), plainMs, sealSqlMs, sealTotalMs, ratio: sealTotalMs / plainMs, mismatch: 0 });
}
