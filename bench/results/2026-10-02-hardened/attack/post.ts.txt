import assert from 'node:assert/strict';import {readFileSync,readdirSync,writeFileSync} from 'node:fs';import {hash} from 'node:crypto';
import {norm,shuffled,ring,scope} from '../../bench/competitor-sim/models.js';import {Codec,positions} from '../../bench/attack-extra/codec.js';import {compileStampQuery} from '../../src/core/stamp-query.js';import {counts} from '../../bench/competitor-sim/attacks.js';import {inferQueryLabels,positionGuesses} from '../../bench/competitor-sim/observation.js';
const out='.local/hardened-attack',fixture=readdirSync('.local/research').find(f=>f.endsWith('-fixture.json'))!,data=shuffled(JSON.parse(readFileSync(`.local/research/${fixture}`,'utf8'))),victim=data.slice(0,10000),reference=data.slice(10000,20000),results:any[]=[];
for(const field of (process.env.POST_FIELDS??'company,address,name,phone').split(',')){
 const raw=victim.map((r:any)=>r[field]),truth=raw.map(norm),refRaw=reference.map((r:any)=>r[field]),ref=refRaw.map(norm),codec=new Codec(field),snapshots=raw.map((v:string,i:number)=>codec.snapshot(v,hash('sha256',`hardened/${field}/${victim[i].id}`,'buffer').subarray(0,16))),keys=new Set<string>();
 for(const value of shuffled(refRaw,108029).slice(0,1000)){const p=await compileStampQuery(ring,codec.p,scope,{op:'contains',value});for(const k of p.keys)keys.add(Buffer.from(k).toString('hex'));}
 const patterns:any[]=truth.map(()=>[]),opaque:any[]=[];for(const key of keys){const loc:any[]=[];snapshots.forEach((s,i)=>{const ps=positions(s,Buffer.from(key,'hex'));if(ps.length){patterns[i].push(ps);loc.push([i,ps]);}});opaque.push({key,locations:loc});}
 // Predictor receives only public reference and anonymous equality-position groups, not key labels or truth.
 const freq=counts(ref),dictionary=[...freq].sort((a,b)=>b[1]-a[1]||a[0].localeCompare(b[0])).map(([v])=>({v,cs:Array.from(v)})),byN=new Map<number,typeof dictionary>();for(const e of dictionary){const xs=byN.get(e.cs.length)??[];xs.push(e);byN.set(e.cs.length,xs);}
 const chosen:string[]=[],sizes:number[]=[],includes:boolean[]=[];for(let i=0;i<10000;i++){
  const candidates=field==='phone'?[]:(byN.get(snapshots[i].n)??[]).filter(e=>{const used=new Set<string>();for(const group of patterns[i]){const pair=e.cs.slice(group[0],group[0]+2).join('');if(used.has(pair)||!group.every((p:number)=>e.cs.slice(p,p+2).join('')===pair))return false;used.add(pair);}return true;});chosen.push(candidates[0]?.v??'');sizes.push(candidates.length);includes.push(candidates.some(e=>e.v===truth[i]));
 }
 const correct=chosen.filter((v,i)=>v===truth[i]).length,metric={field,attack:'D-observe1000-unknown-pattern-dictionary',rows:10000,correct,valuePct:correct/100,candidateIncludesPct:includes.filter(Boolean).length/100,meanCandidateSize:sizes.reduce((a,b)=>a+b,0)/10000,observedKeys:keys.size,method:'reference dictionary + anonymous proof-location pair equality; query plaintext never provided',candidateMeaning:'dictionary candidates only; out-of-dictionary values unresolved'};
 writeFileSync(`${out}/${field}-unknown-pattern.json`,JSON.stringify({metric,predictions:chosen,candidateIncludes:includes,candidateSizes:sizes}));writeFileSync(`${out}/${field}-opaque-locations.json`,JSON.stringify(opaque));results.push(metric);writeFileSync(`${out}/post-results.json`,JSON.stringify(results,null,2));console.log(JSON.stringify(metric));
 const pf=new Map<string,{docs:number,pos:number[],occ:number}>();for(const value of ref){const cs=Array.from(value),seen=new Set<string>();for(let p=0;p+1<cs.length;p++){const pair=cs.slice(p,p+2).join(''),v=pf.get(pair)??{docs:0,pos:[],occ:0};v.pos[p]=(v.pos[p]??0)+1;v.occ++;if(!seen.has(pair))v.docs++;seen.add(pair);pf.set(pair,v);}}
 const observed=opaque.map(o=>{const pos:number[]=[];let occ=0;for(const [,ps]of o.locations)for(const p of ps){pos[p]=(pos[p]??0)+1;occ++;}return {opaque:o.key,locations:new Map(o.locations),docs:o.locations.length,occurrences:occ,pos};}),labels=new Map<string,string>(),available=new Map(pf);
 for(const o of [...observed].sort((a,b)=>b.docs-a.docs)){let best='',distance=Infinity;for(const [pair,f]of available){let d=.1*(Math.log(o.docs+.5)-Math.log(f.docs+.5))**2+.1*(Math.log(o.occurrences+.5)-Math.log(f.occ+.5))**2;for(let p=0;p<Math.max(o.pos.length,f.pos.length);p++)d+=(Math.sqrt((o.pos[p]??0)/(o.occurrences||1))-Math.sqrt((f.pos[p]??0)/(f.occ||1)))**2;if(d<distance){distance=d;best=pair;}}if(best){labels.set(o.opaque,best);available.delete(best);}}
 const partial=positionGuesses(snapshots.map(s=>s.n),observed,labels),posPred=partial.map(ps=>ps.map(c=>c??'?').join('')),posInc=posPred.map((v,i)=>v===truth[i]),posMetric={field,attack:'D-observe1000-unknown-position-frequency',rows:10000,correct:posInc.filter(Boolean).length,valuePct:posInc.filter(Boolean).length/100,candidateIncludesPct:partial.filter((ps,i)=>ps.every((c,j)=>c===undefined||c===Array.from(truth[i])[j])).length/100,meanCandidateSize:null,method:'opaque key full positional distribution + document/occurrence frequency; no query labels',candidateMeaning:'one inferred mapping; all other assignments unresolved'};writeFileSync(`${out}/${field}-unknown-position-frequency.json`,JSON.stringify({metric:posMetric,predictions:posPred,candidateIncludes:posInc,candidateSizes:posPred.map(()=>0)}));results.push(posMetric);writeFileSync(`${out}/post-results.json`,JSON.stringify(results,null,2));console.log(JSON.stringify(posMetric));
 // Exact observation unknown-query frequency assignment: exact hits can be regenerated from opaque equality classes because salted equality proofs are exact.
 const q=shuffled(refRaw,108029).slice(0,1000),exactObs=[...new Set(q.map(norm))].map((label,j)=>({opaque:String(j),hits:truth.flatMap((v,i)=>v===label?[i]:[])}));
 const guesses=inferQueryLabels(ref,exactObs,[...new Set(ref)],(v,q)=>v===q,counts(ref)),prediction=truth.map(()=>''),inc=truth.map(()=>false),sz=truth.map(()=>0);for(const obs of guesses)for(const i of obs.hits){prediction[i]=obs.label??'';inc[i]=prediction[i]===truth[i];sz[i]=1;}
 const exactCorrect=prediction.filter((v,i)=>v===truth[i]).length,exactMetric={field,model:'S0-exact',attack:'D-observe1000-unknown',rows:10000,correct:exactCorrect,valuePct:exactCorrect/100,candidateIncludesPct:inc.filter(Boolean).length/100,meanCandidateSize:sz.reduce((a,b)=>a+b,0)/10000,method:'opaque salted exact-proof hit classes + reference frequency; no query labels',candidateMeaning:'one inferred label per hit class, unobserved unresolved'};writeFileSync(`${out}/${field}-exact-unknown.json`,JSON.stringify({metric:exactMetric,predictions:prediction,candidateIncludes:inc,candidateSizes:sz}));results.push(exactMetric);writeFileSync(`${out}/post-results.json`,JSON.stringify(results,null,2));console.log(JSON.stringify(exactMetric));
}
