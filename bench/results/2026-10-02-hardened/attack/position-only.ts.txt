import {readFileSync,readdirSync,writeFileSync,existsSync} from 'node:fs';import {norm,shuffled} from '../../bench/competitor-sim/models.js';import {positionGuesses} from '../../bench/competitor-sim/observation.js';
const out='.local/hardened-attack',f=readdirSync('.local/research').find(f=>f.endsWith('-fixture.json'))!,data=shuffled(JSON.parse(readFileSync(`.local/research/${f}`,'utf8'))),results:any[]=[];
for(const field of ['company','address','phone','name']){const path=`${out}/${field}-opaque-locations.json`;if(!existsSync(path))continue;const ref=data.slice(10000,20000).map((r:any)=>norm(r[field])),truth=data.slice(0,10000).map((r:any)=>norm(r[field])),opaque=JSON.parse(readFileSync(path,'utf8'));
 const pf=new Map<string,{docs:number,pos:number[],occ:number}>();for(const value of ref){const cs=Array.from(value),seen=new Set<string>();for(let p=0;p+1<cs.length;p++){const pair=cs.slice(p,p+2).join(''),v=pf.get(pair)??{docs:0,pos:[],occ:0};v.pos[p]=(v.pos[p]??0)+1;v.occ++;if(!seen.has(pair))v.docs++;seen.add(pair);pf.set(pair,v);}}
 const observed=opaque.map((o:any)=>{const pos:number[]=[];let occ=0;for(const [,ps]of o.locations)for(const p of ps){pos[p]=(pos[p]??0)+1;occ++;}return {opaque:o.key,locations:new Map<number,number[]>(o.locations),docs:o.locations.length,occurrences:occ,pos};}),labels=new Map<string,string>(),available=new Map(pf);
 for(const o of [...observed].sort((a,b)=>b.docs-a.docs)){let best='',distance=Infinity;for(const [pair,f]of available){let d=.1*(Math.log(o.docs+.5)-Math.log(f.docs+.5))**2+.1*(Math.log(o.occurrences+.5)-Math.log(f.occ+.5))**2;for(let p=0;p<Math.max(o.pos.length,f.pos.length);p++)d+=(Math.sqrt((o.pos[p]??0)/(o.occurrences||1))-Math.sqrt((f.pos[p]??0)/(f.occ||1)))**2;if(d<distance){distance=d;best=pair;}}if(best){labels.set(o.opaque,best);available.delete(best);}}
 const partial=positionGuesses(truth.map(s=>Array.from(s).length),observed,labels),predictions=partial.map(ps=>ps.map(c=>c??'?').join('')),candidateIncludes=partial.map((ps,i)=>ps.every((c,j)=>c===undefined||c===Array.from(truth[i])[j])),metric={field,attack:'D-observe1000-unknown-position-frequency',rows:10000,correct:predictions.filter((v,i)=>v===truth[i]).length,valuePct:predictions.filter((v,i)=>v===truth[i]).length/100,candidateIncludesPct:candidateIncludes.filter(Boolean).length/100,meanCandidateSize:null,method:'anonymous full-position histogram plus document/occurrence frequency; no query labels',candidateMeaning:'one mapping hypothesis; unenumerated mappings unresolved'};writeFileSync(`${out}/${field}-unknown-position-frequency.json`,JSON.stringify({metric,predictions,candidateIncludes,candidateSizes:truth.map(()=>0)}));results.push(metric);console.log(JSON.stringify(metric));}
writeFileSync(`${out}/position-results.json`,JSON.stringify(results,null,2));
