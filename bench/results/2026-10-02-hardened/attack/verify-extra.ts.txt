import assert from 'node:assert/strict';import {readFileSync,readdirSync,writeFileSync} from 'node:fs';import {norm,shuffled} from '../../bench/competitor-sim/models.js';
const out='.local/hardened-attack',f=readdirSync('.local/research').find(f=>f.endsWith('-fixture.json'))!,data=shuffled(JSON.parse(readFileSync(`.local/research/${f}`,'utf8'))).slice(0,10000),checks:any[]=[];
for(const file of readdirSync(out).filter(f=>/-(unknown-pattern|unknown-position-frequency|exact-unknown)\.json$/.test(f)||/^name-graph-(B|C).*\.json$/.test(f))){const a=JSON.parse(readFileSync(`${out}/${file}`,'utf8')),m=a.metric,truth=data.map((r:any)=>norm(r[m.field])),excluded=new Set(a.excluded??[]),correct=a.predictions.filter((v:string,i:number)=>!excluded.has(i)&&v===truth[i]).length;assert.equal(correct,m.correct);assert.equal(10000-excluded.size,m.rows);checks.push({file,correct,rows:m.rows});}
writeFileSync(`${out}/verification-extra.json`,JSON.stringify({passed:true,checks},null,2));console.log(JSON.stringify({passed:true,checked:checks.length}));
